# email-rag 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建 IMAP 邮件全量拉取 → 完全/近似去重 → RAG 友好 Markdown 导出的 CLI 工具。

**Architecture:** 四阶段幂等流水线（fetch/parse/dedup/export），状态存 SQLite，各阶段可独立重跑；近似去重用自研 SimHash-64，LLM 复核可插拔可选。

**Tech Stack:** Python ≥3.10、imapclient、html2text、pyyaml、python-dotenv、pytest（SimHash 自研，禁止新增 near-dup 第三方库）

**Spec:** `docs/superpowers/specs/2026-10-10-email-rag-design.md`

## Global Constraints

- Python ≥3.10，Windows 兼容，全部路径用 `pathlib`，全部文件 UTF-8
- 运行时依赖仅限：`imapclient>=3.3`、`html2text>=2024.2.26`、`pyyaml>=6.0`、`python-dotenv>=1.0`；开发依赖 `pytest>=8.0`
- TDD：每个实现任务先写失败测试再实现；每任务末尾必须 `git commit`
- 提交信息遵循 conventional commits（`feat:` / `fix:` / `test:` / `docs:` / `refactor:`），与仓库历史一致
- 附件永不落盘：解析器跳过所有 `Content-Disposition: attachment` 部分
- 凭证只从 `.env` 读取，任何日志/异常消息不得包含密码
- 数据库写操作必须走事务（`with self._conn:`）

```python
from __future__ import annotations

import os
from dataclasses import dataclass, field
from pathlib import Path

import yaml
from dotenv import load_dotenv


@dataclass
class ImapConfig:
    host: str = ""
    port: int = 993
    user: str = ""
    password: str = ""


@dataclass
class LLMConfig:
    provider: str = "none"  # none | deepseek | openai
    api_key: str = ""
    model: str = ""
    base_url: str = ""


@dataclass
class FetchConfig:
    folders: str = "all"  # "all" 或逗号分隔列表
    exclude: list = field(default_factory=lambda: ["Junk", "Trash", "垃圾邮件", "已删除"])
    rate_limit_per_sec: float = 2.0
    reconnect_max: int = 5


@dataclass
class DedupConfig:
    auto_merge_threshold: float = 0.95
    review_threshold: float = 0.75
    llm_batch_size: int = 20


@dataclass
class ExportConfig:
    output_dir: str = "output"


@dataclass
class AppConfig:
    imap: ImapConfig
    llm: LLMConfig
    fetch: FetchConfig = field(default_factory=FetchConfig)
    dedup: DedupConfig = field(default_factory=DedupConfig)
    export: ExportConfig = field(default_factory=ExportConfig)
    db_path: str = "data/emails.db"
    review_queue_path: str = "data/review_queue.json"


def _override(instance, data: dict):
    """把 dict 中属于该 dataclass 的字段覆盖到实例上。"""
    names = set(instance.__dataclass_fields__)
    for key, value in (data or {}).items():
        if key in names:
            setattr(instance, key, value)
    return instance


def load_config(env_path: str | Path = ".env", config_path: str | Path = "config.yaml") -> AppConfig:
    load_dotenv(dotenv_path=env_path)

    imap = ImapConfig(
        host=os.environ.get("IMAP_HOST", ""),
        port=int(os.environ.get("IMAP_PORT", "993")),
        user=os.environ.get("IMAP_USER", ""),
        password=os.environ.get("IMAP_PASSWORD", ""),
    )
    llm = LLMConfig(
        provider=os.environ.get("LLM_PROVIDER", "none"),
        api_key=os.environ.get("LLM_API_KEY", ""),
        model=os.environ.get("LLM_MODEL", ""),
        base_url=os.environ.get("LLM_BASE_URL", ""),
    )

    fetch, dedup, export = FetchConfig(), DedupConfig(), ExportConfig()
    cfg_file = Path(config_path)
    if cfg_file.exists():
        data = yaml.safe_load(cfg_file.read_text(encoding="utf-8")) or {}
        _override(fetch, data.get("fetch"))
        _override(dedup, data.get("dedup"))
        _override(export, data.get("export"))

    return AppConfig(imap=imap, llm=llm, fetch=fetch, dedup=dedup, export=export)
```

**Interfaces:**
- Consumes: 无
- Produces（后续任务全部依赖这些精确签名）:
  - `Store(db_path)`；`Store.init_schema()`；`Store.close()`
  - `Store.upsert_raw(folder: str, uid: int, mime_bytes: bytes) -> bool`（唯一约束 `(folder, uid)`，重复返回 False）
  - `Store.watermark(folder: str) -> int`（该 folder 已拉取的最大 uid，无则 0）
  - `Store.iter_unparsed_raw(batch: int = 500) -> Iterator[dict]`（dict 含 `id/folder/mime_bytes`；raw 的已解析标记由 `insert_email` 在同一事务中置位，消除崩溃窗口）
  - `Store.insert_email(raw_id: int, folder: str, parsed: dict) -> int`（`parsed` 见 Task 4 `parse_raw` 返回；`to/cc/references` 为 list，入库转 JSON；返回 email id）
  - `Store.get_email(email_id: int) -> dict | None`
  - `Store.update_dedup(email_id: int, status: str, duplicate_of: int | None = None)`
  - `Store.iter_dedup_candidates() -> Iterator[dict]`（`dedup_status='pending'` 的全部 email 行）
  - `Store.finalize_uniques() -> int`（pending → unique，返回数量）
  - `Store.iter_exportable() -> Iterator[dict]`（`status IN ('unique','kept')` 且 `parse_error=''`）
  - `Store.counts() -> dict`（`{"raw": n, "emails": n, "by_status": {...}}`）

`src/email_rag/store.py`:
```python
from __future__ import annotations

import json
import sqlite3
from pathlib import Path
from typing import Iterator

_SCHEMA = """
CREATE TABLE IF NOT EXISTS raw_emails (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    folder TEXT NOT NULL,
    uid INTEGER NOT NULL,
    mime_bytes BLOB NOT NULL,
    parsed INTEGER NOT NULL DEFAULT 0,
    fetched_at TEXT NOT NULL DEFAULT (datetime('now')),
    UNIQUE(folder, uid)
);
CREATE TABLE IF NOT EXISTS emails (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    raw_id INTEGER NOT NULL REFERENCES raw_emails(id),
    folder TEXT NOT NULL,
    subject TEXT NOT NULL DEFAULT '',
    sender TEXT NOT NULL DEFAULT '',
    recipients TEXT NOT NULL DEFAULT '[]',
    cc TEXT NOT NULL DEFAULT '[]',
    date TEXT NOT NULL DEFAULT '',
    message_id TEXT NOT NULL DEFAULT '',
    in_reply_to TEXT NOT NULL DEFAULT '',
    references TEXT NOT NULL DEFAULT '[]',
    body_text TEXT NOT NULL DEFAULT '',
    body_html TEXT NOT NULL DEFAULT '',
    norm_text TEXT NOT NULL DEFAULT '',
    simhash INTEGER NOT NULL DEFAULT 0,
    content_hash TEXT NOT NULL DEFAULT '',
    dedup_status TEXT NOT NULL DEFAULT 'pending',
    duplicate_of INTEGER,
    parse_error TEXT NOT NULL DEFAULT '',
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
);
CREATE INDEX IF NOT EXISTS idx_emails_status ON emails(dedup_status);
CREATE INDEX IF NOT EXISTS idx_emails_mid ON emails(message_id);
CREATE INDEX IF NOT EXISTS idx_emails_hash ON emails(content_hash);
CREATE INDEX IF NOT EXISTS idx_raw_parsed ON raw_emails(parsed);
"""


class Store:
    def __init__(self, db_path: str | Path):
        self._path = Path(db_path)
        self._path.parent.mkdir(parents=True, exist_ok=True)
        self._conn = sqlite3.connect(str(self._path))
        self._conn.row_factory = sqlite3.Row

    def init_schema(self) -> None:
        with self._conn:
            self._conn.executescript(_SCHEMA)

    def close(self) -> None:
        self._conn.close()

    # ---- raw ----
    def upsert_raw(self, folder: str, uid: int, mime_bytes: bytes) -> bool:
        with self._conn:
            cur = self._conn.execute(
                "INSERT OR IGNORE INTO raw_emails(folder, uid, mime_bytes) VALUES (?, ?, ?)",
                (folder, uid, sqlite3.Binary(mime_bytes)),
            )
        return cur.rowcount > 0

    def watermark(self, folder: str) -> int:
        row = self._conn.execute(
            "SELECT MAX(uid) FROM raw_emails WHERE folder = ?", (folder,)
        ).fetchone()
        return int(row[0] or 0)

    def iter_unparsed_raw(self, batch: int = 500) -> Iterator[dict]:
        offset = 0
        while True:
            rows = self._conn.execute(
                "SELECT id, folder, mime_bytes FROM raw_emails WHERE parsed = 0 LIMIT ? OFFSET ?",
                (batch, offset),
            ).fetchall()
            if not rows:
                return
            offset += len(rows)
            for r in rows:
                yield {"id": r["id"], "folder": r["folder"], "mime_bytes": bytes(r["mime_bytes"])}

    # ---- emails ----
    def insert_email(self, raw_id: int, folder: str, parsed: dict) -> int:
        with self._conn:
            cur = self._conn.execute(
                """INSERT INTO emails(raw_id, folder, subject, sender, recipients, cc, date,
                       message_id, in_reply_to, references, body_text, body_html,
                       norm_text, simhash, content_hash, parse_error)
                   VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)""",
                (
                    raw_id, folder, parsed.get("subject", ""), parsed.get("sender", ""),
                    json.dumps(parsed.get("to", []), ensure_ascii=False),
                    json.dumps(parsed.get("cc", []), ensure_ascii=False),
                    parsed.get("date", ""), parsed.get("message_id", ""),
                    parsed.get("in_reply_to", ""),
                    json.dumps(parsed.get("references", []), ensure_ascii=False),
                    parsed.get("body_text", ""), parsed.get("body_html", ""),
                    parsed.get("norm_text", ""), int(parsed.get("simhash", 0)),
                    parsed.get("content_hash", ""), parsed.get("parse_error", ""),
                ),
            )
            self._conn.execute(
                "UPDATE raw_emails SET parsed = 1 WHERE id = ?", (raw_id,)
            )
        return int(cur.lastrowid)

    def get_email(self, email_id: int) -> dict | None:
        row = self._conn.execute("SELECT * FROM emails WHERE id = ?", (email_id,)).fetchone()
        return dict(row) if row else None

    def update_dedup(self, email_id: int, status: str, duplicate_of: int | None = None) -> None:
        with self._conn:
            self._conn.execute(
                "UPDATE emails SET dedup_status = ?, duplicate_of = ? WHERE id = ?",
                (status, duplicate_of, email_id),
            )

    def iter_dedup_candidates(self) -> Iterator[dict]:
        rows = self._conn.execute("SELECT * FROM emails WHERE dedup_status = 'pending'").fetchall()
        for r in rows:
            yield dict(r)

    def finalize_uniques(self) -> int:
        with self._conn:
            cur = self._conn.execute(
                "UPDATE emails SET dedup_status = 'unique' WHERE dedup_status = 'pending'"
            )
        return cur.rowcount

    def iter_exportable(self) -> Iterator[dict]:
        rows = self._conn.execute(
            """SELECT * FROM emails
               WHERE dedup_status IN ('unique', 'kept') AND parse_error = ''"""
        ).fetchall()
        for r in rows:
            yield dict(r)

    def counts(self) -> dict:
        raw = self._conn.execute("SELECT COUNT(*) FROM raw_emails").fetchone()[0]
        emails = self._conn.execute("SELECT COUNT(*) FROM emails").fetchone()[0]
        by_status = {
            row[0]: row[1]
            for row in self._conn.execute(
                "SELECT dedup_status, COUNT(*) FROM emails GROUP BY dedup_status"
            )
        }
        return {"raw": raw, "emails": emails, "by_status": by_status}
```

`tests/test_simhash.py`:

```python
from email_rag.simhash import hamming_distance, similarity, simhash64, tokenize


def test_tokenize_mixed_text():
    assert tokenize("Hello 世界 123") == ["hello", "世", "界", "123"]


def test_simhash_identical_texts_zero_distance():
    text = "通知 关于 会议 改期 明天 上午 十点"
    assert hamming_distance(simhash64(tokenize(text)), simhash64(tokenize(text))) == 0


def test_simhash_similar_texts_small_distance():
    a = simhash64(tokenize("关于会议改期的通知 明天上午十点 会议室A"))
```

```python
from __future__ import annotations

import hashlib
import re

_TOKEN_RE = re.compile(r"[a-z0-9]+|[一-鿿]")


def tokenize(text: str) -> list[str]:
    """英文/数字按词切分，CJK 按单字切分。"""
    return _TOKEN_RE.findall(text.lower())


def _hash64(token: str) -> int:
    digest = hashlib.blake2b(token.encode("utf-8"), digest_size=8).digest()
    return int.from_bytes(digest, "big")


def simhash64(tokens: list[str]) -> int:
    if not tokens:
        return 0
    weights = [0] * 64
    for token in tokens:
        h = _hash64(token)
        for i in range(64):
            weights[i] += 1 if (h >> i) & 1 else -1
    result = 0
    for i in range(64):
        if weights[i] > 0:
            result |= 1 << i
    return result


def hamming_distance(a: int, b: int) -> int:
    return bin(a ^ b).count("1")


def similarity(a: int, b: int) -> float:
    """相似度 = 1 - 汉明距离/64，范围 [0, 1]。"""
    return 1.0 - hamming_distance(a, b) / 64.0
```

```python
from __future__ import annotations

import email
import email.policy
import hashlib
import re
from email.header import decode_header, make_header
from email.message import Message
from email.utils import parsedate_to_datetime

import html2text

from .simhash import simhash64, tokenize

SIGNATURE_RE = re.compile(r"^\s*--\s*$", re.MULTILINE)
SUBJECT_PREFIX_RE = re.compile(
    r"^(re|fw|fwd|回复|答复|转发)\s*[:：]\s*", re.IGNORECASE
)


def decode_str(value: str | None) -> str:
    """解码 RFC 2047 encoded-word 主题/地址头。"""
    if not value:
        return ""
    try:
        return str(make_header(decode_header(value)))
    except Exception:
        return value


def decode_bytes(data: bytes, charset: str | None) -> str:
    candidates = []
    if charset:
        candidates.append(charset)
    candidates += ["utf-8", "gb18030", "latin-1"]
    for cs in candidates:
        try:
            return data.decode(cs)
        except (UnicodeDecodeError, LookupError):
            continue
    return data.decode("utf-8", errors="replace")


def normalize_text(text: str) -> str:
    """去签名块、压缩空白、小写，供去重比较。"""
    sig = SIGNATURE_RE.search(text)
    if sig:
        text = text[: sig.start()]
    text = re.sub(r"\s+", " ", text)
    return text.strip().lower()


def normalize_subject(subject: str) -> str:
    s = decode_str(subject).strip().lower()
    prev = None
    while prev != s:
        prev = s
        s = SUBJECT_PREFIX_RE.sub("", s).strip()
    return s


def _html_to_text(html: str) -> str:
    converter = html2text.HTML2Text()
    converter.body_width = 0
    converter.ignore_images = True
    return converter.handle(html).strip()


def _extract_body(msg: Message) -> tuple[str, str]:
    text_part: Message | None = None
    html_part: Message | None = None
    if msg.is_multipart():
        for part in msg.walk():
            disposition = str(part.get("Content-Disposition", "")).lower()
            if "attachment" in disposition:
                continue  # 附件永不进入正文
            ctype = part.get_content_type()
            if ctype == "text/plain" and text_part is None:
                text_part = part
            elif ctype == "text/html" and html_part is None:
                html_part = part
    else:
        ctype = msg.get_content_type()
        if ctype == "text/plain":
            text_part = msg
        elif ctype == "text/html":
            html_part = msg

    body_text, body_html = "", ""
    if text_part is not None:
        payload = text_part.get_payload(decode=True)
        if payload:
            body_text = decode_bytes(payload, text_part.get_content_charset())
    if html_part is not None:
        payload = html_part.get_payload(decode=True)
        if payload:
            body_html = decode_bytes(payload, html_part.get_content_charset())
    if not body_text and body_html:
        body_text = _html_to_text(body_html)
    return body_text.strip(), body_html.strip()


def parse_mime(mime_bytes: bytes, folder: str) -> dict:
    msg = email.message_from_bytes(mime_bytes, policy=email.policy.default)
    body_text, body_html = _extract_body(msg)

    date_str = ""
    raw_date = msg.get("Date")
    if raw_date:
        try:
            date_str = parsedate_to_datetime(str(raw_date)).isoformat()
        except Exception:
            date_str = str(raw_date)

    message_id = (msg.get("Message-ID") or msg.get("Message-Id") or "").strip()
    references_raw = msg.get("References", "")
    references = references_raw.split() if isinstance(references_raw, str) else []

    return {
        "folder": folder,
        "subject": decode_str(msg.get("Subject", "")),
        "sender": decode_str(msg.get("From", "")),
        "to": [decode_str(x) for x in str(msg.get("To", "")).split(",") if x.strip()],
        "cc": [decode_str(x) for x in str(msg.get("Cc", "")).split(",") if x.strip()],
        "date": date_str,
        "message_id": message_id,
        "in_reply_to": (msg.get("In-Reply-To") or "").strip(),
        "references": references,
        "body_text": body_text,
        "body_html": body_html,
    }


def parse_raw(mime_bytes: bytes, folder: str) -> dict:
    """解析 + 规范化 + 哈希。永不抛异常。"""
    empty = {
        "subject": "", "sender": "", "to": [], "cc": [], "date": "",
        "message_id": "", "in_reply_to": "", "references": [],
        "body_text": "", "body_html": "", "norm_text": "",
        "simhash": 0, "content_hash": "", "parse_error": "",
    }
    try:
        parsed = parse_mime(mime_bytes, folder)
    except Exception as exc:  # noqa: BLE001
        return {**empty, "parse_error": f"{type(exc).__name__}: {exc}"}

    norm = normalize_text(parsed["body_text"] or parsed["subject"])
    return {
        **parsed,
        "norm_text": norm,
        "simhash": simhash64(tokenize(norm)),
        "content_hash": hashlib.sha256(norm.encode("utf-8")).hexdigest(),
        "parse_error": "",
    }
```

```python
from __future__ import annotations

import logging
import time
from typing import Callable

import imapclient

from .config import FetchConfig, ImapConfig
from .store import Store

log = logging.getLogger(__name__)

_BATCH = 500


class FetchError(Exception):
    """folder 重连耗尽后抛出。"""


class Fetcher:
    def __init__(self, imap_cfg: ImapConfig, fetch_cfg: FetchConfig,
                 store: Store, client_factory: Callable | None = None):
        self._imap = imap_cfg
        self._cfg = fetch_cfg
        self._store = store
        self._factory = client_factory or self._default_factory

    def _default_factory(self):
        return imapclient.IMAPClient(
            self._imap.host, port=self._imap.port, ssl=True, timeout=60
        )

    def _login(self, client):
        client.login(self._imap.user, self._imap.password)
        return client

    # ---- folders ----
    def list_folders(self) -> list[str]:
        client = self._login(self._factory())
        try:
            names = []
            for flags, _delim, name in client.list_folders():
                if b"\\Noselect" in flags:
                    continue
                names.append(name)
            return names
        finally:
            client.logout()

    def resolve_folders(self) -> list[str]:
        available = self.list_folders()
        if self._cfg.folders == "all":
            excluded = set(self._cfg.exclude)
            return [f for f in available if f not in excluded]
        wanted = [f.strip() for f in self._cfg.folders.split(",") if f.strip()]
        return [f for f in wanted if f in available]

    # ---- fetch ----
    def fetch_all(self) -> dict[str, int]:
        result = {}
        for folder in self.resolve_folders():
            result[folder] = self.fetch_folder(folder)
        return result

    def fetch_folder(self, folder: str) -> int:
        delay, last_err = 1.0, None
        for attempt in range(self._cfg.reconnect_max):
            try:
                return self._fetch_folder_once(folder)
            except Exception as exc:  # noqa: BLE001
                last_err = exc
                log.warning("fetch %s attempt %d/%d failed: %s",
                            folder, attempt + 1, self._cfg.reconnect_max, exc)
                time.sleep(delay)
                delay *= 2
        raise FetchError(
            f"folder {folder} failed after {self._cfg.reconnect_max} attempts: {last_err}"
        )

    def _fetch_folder_once(self, folder: str) -> int:
        client = self._login(self._factory())
        fetched = 0
        try:
            client.select_folder(folder, readonly=True)
            watermark = self._store.watermark(folder)
            uids = sorted(u for u in client.search() if u > watermark)
            for start in range(0, len(uids), _BATCH):
                batch = uids[start: start + _BATCH]
                data = client.fetch(batch, ["BODY.PEEK[]"])
                for uid in sorted(data):
                    entry = data[uid]
                    mime = entry.get(b"BODY[]") or entry.get("BODY[]")
                    if mime is None:
                        continue
                    if self._store.upsert_raw(folder, int(uid), bytes(mime)):
                        fetched += 1
                if self._cfg.rate_limit_per_sec > 0:
                    time.sleep(len(batch) / self._cfg.rate_limit_per_sec)
            return fetched
        finally:
            client.logout()
```

```python
from __future__ import annotations

import json
import logging
from dataclasses import asdict, dataclass
from datetime import datetime
from itertools import combinations
from pathlib import Path

from .parser import normalize_subject
from .simhash import similarity
from .store import Store

log = logging.getLogger(__name__)

_PREVIEW_LEN = 300
_SENDER_WINDOW_DAYS = 7


@dataclass
class ReviewPair:
    a: int
    b: int
    similarity: float
    subject: str


# ---------- 完全重复 ----------

def exact_dedup(store: Store) -> int:
    groups: dict[tuple, list[dict]] = {}
    for rec in store.iter_dedup_candidates():
        if rec["message_id"]:
            key = ("mid", rec["message_id"].strip().lower())
            groups.setdefault(key, []).append(rec)
        if rec["content_hash"]:
            groups.setdefault(("hash", rec["content_hash"]), []).append(rec)

    marked = 0
    for group in groups.values():
        if len(group) < 2:
            continue
        keeper = max(group, key=lambda r: len(r["body_text"]))
        for rec in group:
            if rec["id"] == keeper["id"]:
                store.update_dedup(rec["id"], "kept")
            else:
                store.update_dedup(rec["id"], "duplicate", keeper["id"])
                marked += 1
    return marked


# ---------- 近似重复 ----------

def _parse_date(value: str):
    try:
        return datetime.fromisoformat(value)
    except (ValueError, TypeError):
        return None


def _days_between(a: str, b: str) -> float:
    da, db = _parse_date(a), _parse_date(b)
    if da is None or db is None:
        return float("inf")
    return abs((da - db).total_seconds()) / 86400.0


def _candidate_pairs(records: list[dict]) -> set[tuple[int, int]]:
    pairs: set[tuple[int, int]] = set()
    by_subject: dict[str, list[dict]] = {}
    by_sender: dict[str, list[dict]] = {}
    for r in records:
        by_subject.setdefault(normalize_subject(r["subject"]), []).append(r)
        by_sender.setdefault(r["sender"], []).append(r)

    for group in by_subject.values():
        for a, b in combinations(group, 2):
            pairs.add((min(a["id"], b["id"]), max(a["id"], b["id"])))

    for group in by_sender.values():
        group.sort(key=lambda r: r["date"])
        for i, a in enumerate(group):
            for b in group[i + 1:]:
                if _days_between(a["date"], b["date"]) > _SENDER_WINDOW_DAYS:
                    break
                pairs.add((min(a["id"], b["id"]), max(a["id"], b["id"])))
    return pairs


def near_dedup(store: Store, auto_threshold: float = 0.95,
               review_threshold: float = 0.75) -> list[ReviewPair]:
    pending = [r for r in store.iter_dedup_candidates()
               if r["norm_text"].strip() and r["simhash"]]
    by_id = {r["id"]: r for r in pending}
    pairs = _candidate_pairs(pending)

    judged = []
    for ida, idb in pairs:
        sim = similarity(by_id[ida]["simhash"], by_id[idb]["simhash"])
        judged.append((ida, idb, sim))
    judged.sort(key=lambda t: -t[2])

    spent: set[int] = set()
    review_pairs: list[ReviewPair] = []
    for ida, idb, sim in judged:
        if ida in spent or idb in spent:
            continue
        if sim >= auto_threshold:
            ra, rb = by_id[ida], by_id[idb]
            keeper = ra if len(ra["body_text"]) >= len(rb["body_text"]) else rb
            other = rb if keeper is ra else ra
            store.update_dedup(keeper["id"], "kept")
            store.update_dedup(other["id"], "duplicate", keeper["id"])
            spent.add(ida)
            spent.add(idb)
        elif sim >= review_threshold:
            ra, rb = by_id[ida], by_id[idb]
            store.update_dedup(ida, "review")
            store.update_dedup(idb, "review")
            spent.add(ida)
            spent.add(idb)
            review_pairs.append(ReviewPair(
                a=ida, b=idb, similarity=round(sim, 4),
                subject=(ra["subject"] or rb["subject"])[:200],
            ))
    return review_pairs


# ---------- 复核队列 ----------

def write_review_queue(store: Store, pairs: list[ReviewPair], path: str | Path) -> None:
    items = []
    for p in pairs:
        ra, rb = store.get_email(p.a), store.get_email(p.b)
        items.append({
            **asdict(p),
            "preview_a": (ra["norm_text"] if ra else "")[:_PREVIEW_LEN],
            "preview_b": (rb["norm_text"] if rb else "")[:_PREVIEW_LEN],
        })
    out = Path(path)
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps({"pairs": items}, ensure_ascii=False, indent=2),
                   encoding="utf-8")


def apply_review(store: Store, queue_path: str | Path) -> int:
    """读取队列中人工/LLM 写入的 verdict：duplicate → 合并，unique → 各自保留。"""
    data = json.loads(Path(queue_path).read_text(encoding="utf-8"))
    applied = 0
    for item in data.get("pairs", []):
        verdict = item.get("verdict")
        if verdict not in ("duplicate", "unique"):
            continue
        a, b = store.get_email(item["a"]), store.get_email(item["b"])
        if not a or not b:
            continue
        if verdict == "unique":
            store.update_dedup(a["id"], "unique")
            store.update_dedup(b["id"], "unique")
        else:
            keeper = a if len(a["body_text"]) >= len(b["body_text"]) else b
            other = b if keeper is a else a
            store.update_dedup(keeper["id"], "kept")
            store.update_dedup(other["id"], "duplicate", keeper["id"])
        applied += 1
    return applied
```

```python
from __future__ import annotations

import json
import logging
import urllib.request

log = logging.getLogger(__name__)

_MAX_TEXT_LEN = 2000  # 每封仅发送截断后的规范化文本片段（spec §12 最小化原则）

_PROMPT_TEMPLATE = """判断两段邮件正文是否为重复内容。
判定为重复：同一内容的不同版本、转发/回复导致的变化、格式差异。
判定为不重复：主题相关但实质内容不同。
仅输出 JSON：{{"duplicate": true}} 或 {{"duplicate": false}}

文本A：
{a}

文本B：
{b}}"""


class LLMError(Exception):
    """LLM 不可用或响应无法解析。"""


def _extract_verdict(content: str) -> bool | None:
    """从响应中提取 {"duplicate": bool}；失败返回 None。"""
    start = content.find("{")
    end = content.rfind("}")
    if start == -1 or end == -1 or end <= start:
        return None
    try:
        data = json.loads(content[start: end + 1])
    except json.JSONDecodeError:
        return None
    if isinstance(data, dict) and isinstance(data.get("duplicate"), bool):
        return data["duplicate"]
    return None


class NullLLM:
    def judge_pair(self, text_a: str, text_b: str) -> bool:
        raise LLMError("LLM_PROVIDER 未配置（none）")


class _ChatCompletionsLLM:
    """共享的 /chat/completions 调用骨架（stdlib urllib，无 SDK 依赖）。"""

    def __init__(self, api_key: str, model: str, base_url: str):
        self._url = f"{base_url.rstrip('/')}/chat/completions"
        self._headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json",
        }
        self._model = model

    def judge_pair(self, text_a: str, text_b: str) -> bool:
        prompt = _PROMPT_TEMPLATE.format(
            a=text_a[:_MAX_TEXT_LEN], b=text_b[:_MAX_TEXT_LEN]
        )
        payload = {
            "model": self._model,
            "messages": [{"role": "user", "content": prompt}],
            "temperature": 0,
            "max_tokens": 60,
        }
        request = urllib.request.Request(
            self._url, data=json.dumps(payload).encode("utf-8"),
            headers=self._headers,
        )
        with urllib.request.urlopen(request, timeout=60) as resp:
            data = json.loads(resp.read().decode("utf-8"))
        content = data["choices"][0]["message"]["content"]
        verdict = _extract_verdict(content)
        if verdict is None:
            raise LLMError(f"无法解析 LLM 响应: {content[:120]}")
        return verdict


class DeepSeekLLM(_ChatCompletionsLLM):
    def __init__(self, api_key: str, model: str = "deepseek-chat",
                 base_url: str = "https://api.deepseek.com"):
        super().__init__(api_key, model or "deepseek-chat", base_url)


class OpenAILLM(_ChatCompletionsLLM):
    def __init__(self, api_key: str, model: str = "gpt-4o-mini",
                 base_url: str = "https://api.openai.com/v1"):
        super().__init__(api_key, model or "gpt-4o-mini", base_url)


def create_llm(provider: str, api_key: str, model: str, base_url: str):
    if provider == "deepseek":
        return DeepSeekLLM(api_key, model, base_url)
    if provider == "openai":
        return OpenAILLM(api_key, model, base_url)
    return NullLLM()
```

```python
def judge_pairs_with_llm(store: Store, pairs: list[ReviewPair], llm,
                         queue_path: str | Path) -> int:
    """逐对 LLM 裁定并写回 verdict；失败的 pair 保持 review 留待人工。"""
    path = Path(queue_path)
    data = json.loads(path.read_text(encoding="utf-8"))
    for item, pair in zip(data.get("pairs", []), pairs):
        if item.get("verdict"):
            continue
        try:
            verdict = llm.judge_pair(item.get("preview_a", ""),
                                     item.get("preview_b", ""))
            item["verdict"] = "duplicate" if verdict else "unique"
        except Exception as exc:  # noqa: BLE001
            log.warning("LLM 裁定失败 pair(%s, %s): %s", pair.a, pair.b, exc)
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2),
                    encoding="utf-8")
    return apply_review(store, path)
```

```python
from __future__ import annotations

import json
import re
import unicodedata
from pathlib import Path

from .store import Store


def slugify(text: str, max_len: int = 40) -> str:
    text = unicodedata.normalize("NFKC", text or "").lower()
    text = re.sub(r"[^\w一-鿿]+", "-", text)
    text = re.sub(r"-{2,}", "-", text).strip("-")
    return (text[:max_len].strip("-") or "untitled")


def render_markdown(record: dict) -> str:
    frontmatter = {
        "subject": record["subject"],
        "from": record["sender"],
        "to": json.loads(record["recipients"] or "[]"),
        "cc": json.loads(record["cc"] or "[]"),
        "date": record["date"],
        "folder": record["folder"],
        "message_id": record["message_id"],
        "labels": [],
        "dedup": record["dedup_status"],
        "content_hash": record["content_hash"][:8],
    }
    lines = ["---"]
    for key, value in frontmatter.items():
        lines.append(f"{key}: {json.dumps(value, ensure_ascii=False)}")
    lines.append("---")
    lines.append("")
    lines.append(record["body_text"])
    return "\n".join(lines) + "\n"


def export_all(store: Store, output_dir: str | Path) -> dict:
    out = Path(output_dir)
    count = 0
    for rec in store.iter_exportable():
        date = rec["date"] or ""
        ym = date[:7] if len(date) >= 7 else "unknown-date"
        day = date[:10].replace("-", "") if len(date) >= 10 else "00000000"
        folder_dir = out / slugify(rec["folder"], max_len=20) / ym
        folder_dir.mkdir(parents=True, exist_ok=True)
        name = f"{day}-{slugify(rec['subject'])}-{rec['content_hash'][:8]}.md"
        (folder_dir / name).write_text(render_markdown(rec), encoding="utf-8")
        count += 1
    return {"exported": count, "output_dir": str(out)}
```

```python
import argparse

def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="email-rag",
                                     description="IMAP 邮件拉取、去重、导出 Markdown")
    parser.add_argument("--env", default=".env", help=".env 路径")
    parser.add_argument("--config", default="config.yaml", help="config.yaml 路径")
    sub = parser.add_subparsers(dest="command", required=True)
    sub.add_parser("fetch", help="IMAP 拉取邮件到 SQLite")
    sub.add_parser("parse", help="解析 raw 邮件为结构化记录")
    dedup_parser = sub.add_parser("dedup", help="去重；可带队列文件导入裁定")
    dedup_parser.add_argument("queue", nargs="?", default=None,
                              help="复核队列 JSON 文件（导入人工/LLM 裁定）")
    dedup_parser.add_argument("--llm", action="store_true", help="用 LLM 自动裁定复核队列")
    sub.add_parser("export", help="导出 Markdown")
    sub.add_parser("report", help="统计报告")
    return parser


def run_parse(store: Store) -> int:
    count = 0
    for raw in store.iter_unparsed_raw():
        parsed = parse_raw(raw["mime_bytes"], raw["folder"])
        store.insert_email(raw["id"], raw["folder"], parsed)
        count += 1
    return count


def run_dedup(store: Store, cfg: AppConfig, use_llm: bool = False) -> None:
    exact = exact_dedup(store)
    pairs = near_dedup(store, cfg.dedup.auto_merge_threshold,
                       cfg.dedup.review_threshold)
    print(f"完全重复合并 {exact} 封；近似重复复核队列 {len(pairs)} 对")
    if pairs:
        write_review_queue(store, pairs, cfg.review_queue_path)
        print(f"复核队列已写入 {cfg.review_queue_path}")
        if use_llm and cfg.llm.provider != "none":
            llm = create_llm(cfg.llm.provider, cfg.llm.api_key,
                             cfg.llm.model, cfg.llm.base_url)
            applied = judge_pairs_with_llm(store, pairs, llm, cfg.review_queue_path)
            print(f"LLM 已裁定 {applied} 对")
        else:
            print("人工裁定后运行：email-rag dedup data/review_queue.json")
    finalized = store.finalize_uniques()
    print(f"标记 unique {finalized} 封")


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    logging.basicConfig(level=logging.INFO,
                        format="%(levelname)s %(name)s: %(message)s")
    cfg = load_config(args.env, args.config)

    if args.command == "fetch":
        if not (cfg.imap.host and cfg.imap.user and cfg.imap.password):
            raise SystemExit("IMAP_HOST / IMAP_USER / IMAP_PASSWORD 未配置（见 .env.example）")
        store = Store(cfg.db_path)
        store.init_schema()
        try:
            fetcher = Fetcher(cfg.imap, cfg.fetch, store)
            for folder, n in fetcher.fetch_all().items():
                print(f"{folder}: 新拉取 {n} 封")
        finally:
            store.close()
        return 0

    if args.command == "dedup" and args.queue:
        store = Store(cfg.db_path)
        store.init_schema()
        try:
            applied = apply_review(store, args.queue)
            print(f"已导入裁定 {applied} 对")
            print(f"标记 unique {store.finalize_uniques()} 封")
        finally:
            store.close()
        return 0

    store = Store(cfg.db_path)
    store.init_schema()
    try:
        if args.command == "parse":
            print(f"解析完成：{run_parse(store)} 封")
        elif args.command == "dedup":
            run_dedup(store, cfg, use_llm=args.llm)
        elif args.command == "export":
            print(json.dumps(export_all(store, cfg.export.output_dir),
                             ensure_ascii=False))
        elif args.command == "report":
            print(json.dumps(store.counts(), ensure_ascii=False, indent=2))
    finally:
        store.close()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
